// Sub-10ms Cached Redirection
export async function resolveShortUrl(code) {
  const cached = await redis.get(`slug:${code}`);
  if (cached) {
    analyticsQueue.push({ code, timestamp: Date.now() });
    return JSON.parse(cached);
  }
  const record = await UrlDAO.findByCode(code);
  await redis.setex(`slug:${code}`, 3600, record.url);
  return record.url;
}

// Git Health & PR Velocity Metrics
export async function calcVelocity(owner, repo) {
  const pulls = await octokit.rest.pulls.list({ owner, repo });
  const leadTimeHours = pulls.data.map(p => 
    (new Date(p.merged_at) - new Date(p.created_at)) / 3.6e6
  );
  return { avgCycleTime: mean(leadTimeHours) };
}
